import sys
import fontforge
TARGET_WIDTH = 1379
MODE = 1  # 0 = Info mode, 1 = Edit mode
font = fontforge.activeFont()
if font is None:
    fontforge.postError("Error", "No active font open in FontForge!")
    sys.exit(1)
log = []
nontrivial_chars = []
nontrivial_chars.extend(range(0x0030, 0x003A))
nontrivial_chars.extend(range(0x0041, 0x005B))
nontrivial_chars.extend(range(0x0061, 0x007B))
nontrivial_widths = []
for codepoint in nontrivial_chars:
    if codepoint in font:
        nontrivial_widths.append(font[codepoint].width)
if nontrivial_widths:
    nontrivial_widths.sort()
    count_less_or_equal = sum(1 for w in nontrivial_widths if w <= TARGET_WIDTH)
    percentile = (count_less_or_equal / len(nontrivial_widths)) * 100
    log.append("--- FONT MONOSPACE REPORT ---")
    log.append(f"Target Width: {TARGET_WIDTH}")
    log.append(f"Percentile of target width among non-trivial glyphs: {percentile:.2f}%")
else:
    log.append("Warning: No non-trivial ASCII glyphs found in this font.")
exceeding_glyphs = []
for glyph in font.glyphs():
    if glyph.width > TARGET_WIDTH:
        excess = glyph.width - TARGET_WIDTH
        if glyph.unicode > 0:
            char_repr = f"U+{glyph.unicode:04X} ('{chr(glyph.unicode)}')"
        else:
            char_repr = f"Glyph Name: '{glyph.glyphname}'"
        exceeding_glyphs.append((char_repr, glyph.width, excess))
log.append(f"\nGlyphs exceeding chosen width ({len(exceeding_glyphs)} found):")
for char_repr, width, excess in exceeding_glyphs:
    log.append(f"  - {char_repr} | Current Width: {width} | Excess: +{excess}")
if MODE == 1:
    log.append("\n--- EDIT MODE: Centering glyphs (Foreground only) and applying target width ---")
    modified_count = 0
    for glyph in font.glyphs():
        fore_layer = glyph.layers["Fore"]
        bbox = fore_layer.boundingBox()
        xmin, xmax = bbox[0], bbox[2]
        glyph_visual_width = xmax - xmin
        if glyph_visual_width > 0:
            desired_lb = (TARGET_WIDTH - glyph_visual_width) / 2.0
            shift_x = desired_lb - xmin
            glyph.transform((1, 0, 0, 1, shift_x, 0))
        glyph.width = TARGET_WIDTH
        modified_count += 1
    log.append(f"Successfully monospaced {modified_count} glyphs to width {TARGET_WIDTH}.")
report_text = "\n".join(log)
if fontforge.hasUserInterface():
    fontforge.postNotice("Monospace Script Output", report_text)
else:
    print(report_text)
