// a solution by Infinity Integral

// This is written for Informatics Olympiad

// Raffles Institution Novcourse Final Contest 2023 question "sunshine"

#include <bits/stdc++.h>
using namespace std;
#define int long long
#define z128 __int128

inline int readInt(){
    int x = 0;
    char ch = getchar();
    while(ch < '0' || ch > '9'){
        ch = getchar_unlocked();
    }
    while(ch >= '0' && ch <= '9'){
        x = (x << 3) + (x << 1) + ch - '0';
        ch = getchar_unlocked();
    }
    return x;
}

int n;
int q;
int k;
int w[303];
int v[303];
int xw;
int xv;
int dp[303][5005];

signed main(){
    ios_base::sync_with_stdio(false); cin.tie(0); cout.tie(0);
    cin >> n >> q >> k;
    for(int i=1; i<=n; i++){
        cin >> w[i];
        cin >> v[i];
    }
    for(int i=1; i<=n; i++){
        for(int j=1; j<=k; j++){
            if(j - w[i] < 0){dp[i][j] = max(dp[i-1][j], dp[i][j-1]);}
            else{dp[i][j] = max(dp[i-1][j-w[i]]+v[i], max(dp[i-1][j], dp[i][j-1]));}
        }
    }
    for(int i=0; i<q; i++){
        cin >> xw >> xv;
        if(k >= xw && dp[n][k-xw] + xv > dp[n][k]){cout << "YES\n";}
        else{cout << "NO\n";}
    }
}
